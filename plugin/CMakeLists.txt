cmake_minimum_required(VERSION 3.21)
project(AimpKaraokeStudio LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32)
  message(FATAL_ERROR "The AIMP plugin requires Windows and MSVC")
endif()
if(NOT DEFINED AIMP_SDK_DIR OR NOT EXISTS "${AIMP_SDK_DIR}/apiPlugin.h")
  message(FATAL_ERROR "Set -DAIMP_SDK_DIR=C:/path/to/AIMP_SDK/Sources/Cpp (apiPlugin.h must exist)")
endif()
add_library(AimpKaraokeStudio SHARED Plugin.cpp PipeServer.cpp Lrc.cpp BatchLyrics.cpp)
target_include_directories(AimpKaraokeStudio PRIVATE "${AIMP_SDK_DIR}" .)
target_compile_definitions(AimpKaraokeStudio PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(AimpKaraokeStudio PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(AimpKaraokeStudio PRIVATE shell32 user32 advapi32 winhttp)
set_target_properties(AimpKaraokeStudio PROPERTIES OUTPUT_NAME "AimpKaraokeStudio" PREFIX "")
